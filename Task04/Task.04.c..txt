#include <stdio.h>

#define RUPEES_TO_PAISE(r) ((long long)(r) * 100)
#define SQUARE(x) ((x) * (x))
#define MAX(a, b) ((a) > (b) ? (a) : (b))

static inline int square_function(int x)
{
    return x * x;
}

int main(void)
{
    int a = 4;
    int b = 7;

    printf("5 rupees in paise = %lld\n", RUPEES_TO_PAISE(5));
    printf("Square of a + 1 = %d\n", SQUARE(a + 1));
    printf("Maximum value = %d\n", MAX(a, b));
    printf("Square using function = %d\n", square_function(a));

    return 0;
}
